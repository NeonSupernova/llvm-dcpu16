cmake_minimum_required(VERSION 3.20)
project(DCPU16Modern LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(LLVM_VERSION_MAJOR LESS 22 OR LLVM_VERSION_MAJOR GREATER 23)
  message(FATAL_ERROR "Supported LLVM development kits: 22 and 23; found ${LLVM_PACKAGE_VERSION}")
endif()
message(STATUS "Building DCPU16 against LLVM ${LLVM_PACKAGE_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${LLVM_CMAKE_DIR}")
include(AddLLVM)
find_program(DCPU16_TABLEGEN NAMES llvm-tblgen HINTS "${LLVM_TOOLS_BINARY_DIR}" REQUIRED NO_DEFAULT_PATH)
set(DCPU16_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lib/DCPU16")
set(DCPU16_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${DCPU16_GENERATED}")
file(GLOB DCPU16_TD CONFIGURE_DEPENDS "${DCPU16_DIR}/*.td")
set(DCPU16_TD_INCLUDES -I "${DCPU16_DIR}")
foreach(include_dir IN LISTS LLVM_INCLUDE_DIRS)
  list(APPEND DCPU16_TD_INCLUDES -I "${include_dir}")
endforeach()
function(dcpu16_tablegen output generator)
  add_custom_command(OUTPUT "${DCPU16_GENERATED}/${output}"
    COMMAND "${DCPU16_TABLEGEN}" "-${generator}" ${DCPU16_TD_INCLUDES} "${DCPU16_DIR}/DCPU16.td"
      -o "${DCPU16_GENERATED}/${output}"
    DEPENDS ${DCPU16_TD} VERBATIM)
  set(DCPU16_GENERATED_FILES ${DCPU16_GENERATED_FILES} "${DCPU16_GENERATED}/${output}" PARENT_SCOPE)
endfunction()
dcpu16_tablegen(DCPU16GenRegisterInfo.inc gen-register-info)
dcpu16_tablegen(DCPU16GenInstrInfo.inc gen-instr-info)
dcpu16_tablegen(DCPU16GenAsmWriter.inc gen-asm-writer)
dcpu16_tablegen(DCPU16GenDAGISel.inc gen-dag-isel)
dcpu16_tablegen(DCPU16GenCallingConv.inc gen-callingconv)
dcpu16_tablegen(DCPU16GenSubtargetInfo.inc gen-subtarget)
add_custom_target(DCPU16TableGen DEPENDS ${DCPU16_GENERATED_FILES})
file(GLOB_RECURSE DCPU16_SOURCES CONFIGURE_DEPENDS "${DCPU16_DIR}/*.cpp")
add_library(DCPU16CodeGen STATIC ${DCPU16_SOURCES})
add_dependencies(DCPU16CodeGen DCPU16TableGen)
target_include_directories(DCPU16CodeGen PUBLIC "${DCPU16_DIR}" "${DCPU16_GENERATED}")
target_include_directories(DCPU16CodeGen SYSTEM PUBLIC ${LLVM_INCLUDE_DIRS})
string(REPLACE ";" " " DCPU16_LLVM_DEFINITION_STRING "${LLVM_DEFINITIONS}")
separate_arguments(DCPU16_LLVM_DEFINITIONS NATIVE_COMMAND "${DCPU16_LLVM_DEFINITION_STRING}")
target_compile_options(DCPU16CodeGen PUBLIC ${DCPU16_LLVM_DEFINITIONS})
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(DCPU16CodeGen PUBLIC -fno-rtti)
endif()
if(LLVM_ENABLE_ASSERTIONS)
  target_compile_options(DCPU16CodeGen PUBLIC -UNDEBUG)
endif()
add_executable(dcpu16-llc tools/dcpu16-llc.cpp)
target_link_libraries(dcpu16-llc PRIVATE DCPU16CodeGen LLVM)
# Bootstrapped LLVM can live inside this source tree, which the automatic
# link-path property excludes. Preserve that SDK location explicitly too.
set_target_properties(dcpu16-llc PROPERTIES
  INSTALL_RPATH "${LLVM_LIBRARY_DIRS}"
  INSTALL_RPATH_USE_LINK_PATH TRUE)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(DCPU16_CLANG_EXECUTABLE "" CACHE FILEPATH "Optional Clang with MSP430 support for C frontend tests")
foreach(level IN ITEMS 0 2)
  add_test(NAME dcpu16-execution-O${level}
    COMMAND "${CMAKE_COMMAND}" -E env
      "DCPU16_LLC=$<TARGET_FILE:dcpu16-llc>" "DCPU16_OPT_LEVEL=${level}" "DCPU16_CLANG=${DCPU16_CLANG_EXECUTABLE}"
      "DCPU16_BUILD_DIR=${CMAKE_BINARY_DIR}"
      "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -v)
endforeach()
install(TARGETS dcpu16-llc RUNTIME DESTINATION bin)
install(PROGRAMS tools/dcpu16.py DESTINATION share/llvm-dcpu16)
install(FILES tools/dcpu16_object.py DESTINATION share/llvm-dcpu16)
