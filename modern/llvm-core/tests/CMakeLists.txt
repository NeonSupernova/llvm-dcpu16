cmake_minimum_required(VERSION 3.20)
project(DCPU16WordLayoutTests LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "The experimental core patch requires LLVM 23.1.4")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dcpu16-word-layout word-layout.cpp)
target_include_directories(dcpu16-word-layout SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
string(REPLACE ";" " " LLVM_DEFINITION_STRING "${LLVM_DEFINITIONS}")
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND "${LLVM_DEFINITION_STRING}")
target_compile_options(dcpu16-word-layout PRIVATE ${LLVM_DEFINITIONS_LIST})
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(dcpu16-word-layout PRIVATE -fno-rtti)
endif()
# Static components keep this experiment isolated from the ordinary LLVM SDK.
target_link_libraries(dcpu16-word-layout PRIVATE LLVMAsmParser LLVMBitReader LLVMBitWriter LLVMAnalysis LLVMCore)
enable_testing()
add_test(NAME dcpu16-word-layout COMMAND dcpu16-word-layout)
add_executable(dcpu16-optimizer-memory optimizer-memory.cpp)
target_include_directories(dcpu16-optimizer-memory SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_options(dcpu16-optimizer-memory PRIVATE ${LLVM_DEFINITIONS_LIST} -fno-rtti)
target_link_libraries(dcpu16-optimizer-memory PRIVATE LLVMTransformUtils LLVMInterpreter LLVMAsmParser LLVMAnalysis LLVMCore)
add_test(NAME dcpu16-native-coercion COMMAND dcpu16-optimizer-memory coercion)
add_test(NAME dcpu16-native-loops COMMAND dcpu16-optimizer-memory loops)
